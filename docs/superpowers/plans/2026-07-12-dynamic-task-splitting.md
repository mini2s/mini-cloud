# Dynamic Task Splitting Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 workflow 新增"任务拆分节点"（Split Node），由 Agent 根据上下文智能生成子任务列表，经人审核后批量创建子 issue，各自绑定独立 workflow 并行执行。

**Architecture:** 新增 `kind: "split"` 节点类型、`multica_workflow_split_task` 表、`SplitOrchestrator` Go 服务、4 个 API 端点（generate/approve/tasks/cancel）、6 个前端组件、3 个预检项。拆分节点在 `WorkflowService` 状态机之外独立编排，通过 `OnNodeRunTransition` hook 接入现有管线。

**Tech Stack:** Go (Chi + sqlc + pgx), TypeScript (React + ReactFlow + TanStack Query + Zustand), PostgreSQL

**Revision Note (v0.2.0):** This plan has been corrected against the current repository contracts. Implementers must not copy snippets blindly if generated sqlc types differ after `make sqlc`; update the snippet and this plan together.

## Global Constraints

- 子任务嵌套限制为两层（父→子），`max_concurrency` 默认 5，`max_failures` 默认 0
- barrier 模式等所有子任务终态，pipeline 模式子 issue 创建后立即释放下游
- 子任务间支持 DAG 依赖关系（串行/并行），按拓扑顺序 + max_concurrency 调度
- 父节点取消时级联停止所有子任务，带防呆确认
- 新 API response schema 必须在 `packages/core/api/schemas.ts` 使用 zod + `parseWithFallback`，不允许裸 `as` cast
- 所有 Go handler 必须通过 `nodeRunID` 定位 split，先加载 node_run 再反查 workflow_run → workflow → workspace 做权限校验
- `multica_issue.origin_type` CHECK 约束必须新增 `workflow_split`，并必须保留现有 `autopilot`、`quick_create`、`workflow` 值；不要新增 handler 不支持的 `manual`
- API 路径使用扁平 `/api/node-runs/{nodeRunId}/split/*` 格式
- 中文注释/文案使用约定词汇表（见 `apps/docs/content/docs/developers/conventions.zh.mdx`）
- Go/TypeScript 代码注释必须使用英文；中文只允许出现在用户可见文案和本文档说明中
- split Agent 完成回调必须接入现有 `TaskService.OnTaskCompleted -> WorkflowService.HandleWorkflowTaskCompletion` 链路，否则拆分结果不会进入审核

```sql
-- Dynamic task splitting: stores Agent-generated sub-task proposals
-- and their lifecycle after human review and issue materialization.
CREATE TABLE IF NOT EXISTS multica_workflow_split_task (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    node_run_id   UUID NOT NULL REFERENCES multica_workflow_node_run(id) ON DELETE CASCADE,
    workspace_id  UUID NOT NULL REFERENCES multica_workspace(id) ON DELETE CASCADE,

    -- Split plan content generated by Agent or edited by a user.
    title         TEXT NOT NULL,
    description   TEXT NOT NULL DEFAULT '',
    suggested_assignee_type TEXT,  -- 'member' | 'agent' | 'squad'
    suggested_assignee_id   UUID,
    depends_on    JSONB NOT NULL DEFAULT '[]',  -- ["<split_task_id>", ...]
    sort_order    INT NOT NULL DEFAULT 0,

    -- Lifecycle status.
    status        TEXT NOT NULL DEFAULT 'draft' CHECK (status IN (
        'draft',
        'approved',
        'discarded',
        'created',
        'running',
        'done',
        'failed',
        'cancelled',
        'skipped'
    )),

    -- Backfilled after issue/run creation.
    issue_id      UUID REFERENCES multica_issue(id) ON DELETE SET NULL,
    run_id        UUID REFERENCES multica_workflow_run(id) ON DELETE SET NULL,

    created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_workflow_split_task_node_run ON multica_workflow_split_task(node_run_id);
CREATE INDEX IF NOT EXISTS idx_workflow_split_task_issue ON multica_workflow_split_task(issue_id);
CREATE INDEX IF NOT EXISTS idx_workflow_split_task_run ON multica_workflow_split_task(run_id);

-- Extend origin_type CHECK on multica_issue to accept workflow_split.
-- The constraint name is preserved from pre-prefix migrations.
ALTER TABLE multica_issue DROP CONSTRAINT IF EXISTS issue_origin_type_check;
ALTER TABLE multica_issue ADD CONSTRAINT issue_origin_type_check
    CHECK (origin_type IN ('autopilot', 'quick_create', 'workflow', 'workflow_split'));
```

```go
// HandleSplitNode is the entry point called when a kind=split node run
// transitions. It dispatches to the correct handler based on the target status.
func (s *SplitOrchestrator) HandleSplitNode(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
	newStatus string,
) error {
	switch newStatus {
	case NodeRunStatusSplitting:
		return s.handleActivateSplit(ctx, nodeRun)
	case NodeRunStatusAwaitingSplitReview:
		// This transition is driven by Agent task completion callback.
		// No-op here; the callback triggers HandleAgentTaskCompletion directly.
		return nil
	case NodeRunStatusSplitActive:
		// This transition is driven by ApproveSplit.
		return nil
	case NodeRunStatusFailed:
		// Mark all pending split tasks as failed.
		return s.failAllPendingTasks(ctx, nodeRun.ID)
	case NodeRunStatusCancelled:
		return s.cancelSplitNode(ctx, nodeRun)
	default:
		return fmt.Errorf("split orchestrator: unhandled transition %s → %s", nodeRun.Status, newStatus)
	}
}
```

```go
func (s *SplitOrchestrator) handleActivateSplit(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
) error {
	// 1. Read split_config from the workflow node's format_schema.
	node, err := s.Queries.GetWorkflowNode(ctx, nodeRun.WorkflowNodeID)
	if err != nil {
		return fmt.Errorf("get node: %w", err)
	}

	var formatSchema struct {
		SplitConfig SplitConfig `json:"split_config"`
	}
	if err := json.Unmarshal(node.FormatSchema, &formatSchema); err != nil {
		return fmt.Errorf("parse format_schema: %w", err)
	}
	cfg := formatSchema.SplitConfig

	// 2. Validate sub_template_id exists and is active.
	subTplID, err := util.ParseUUID(cfg.SubTemplateID)
	if err != nil {
		return fmt.Errorf("invalid sub_template_id: %w", err)
	}
	subTpl, err := s.Queries.GetWorkflow(ctx, subTplID)
	if err != nil {
		return fmt.Errorf("sub-template not found: %w", err)
	}
	if subTpl.Status != "active" {
		return fmt.Errorf("sub-template %s is not active", subTpl.Title)
	}

	// 3. Nesting guard: verify sub-template has NO kind=split nodes.
	subNodes, err := s.Queries.ListWorkflowNodes(ctx, subTplID)
	if err != nil {
		return fmt.Errorf("list sub-template nodes: %w", err)
	}
	for _, sn := range subNodes {
		var snFormat struct {
			Type string `json:"type"`
		}
		_ = json.Unmarshal(sn.FormatSchema, &snFormat)
		if snFormat.Type == "split" {
			return fmt.Errorf(
				"sub-template %q contains a split node %q — nesting beyond 2 levels is not allowed",
				subTpl.Title, sn.Title,
			)
		}
	}

	// 4. Idempotency: check if split tasks already exist from a previous attempt.
	existing, err := s.Queries.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
	if err == nil && len(existing) > 0 {
		// Already generated — transition directly to awaiting review.
		slog.Info("split: tasks already exist, reusing", "node_run_id", util.UUIDToString(nodeRun.ID))
		_, err := s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusAwaitingSplitReview)
		return err
	}

	// 5. Build Agent prompt and dispatch.
	run, err := s.Queries.GetWorkflowRun(ctx, nodeRun.WorkflowRunID)
	if err != nil {
		return fmt.Errorf("get run: %w", err)
	}
	issue, err := s.findIssueForRun(ctx, run)
	if err != nil {
		return fmt.Errorf("find issue for run: %w", err)
	}

	// Gather upstream node run outputs as context.
	upstreamOutputs, err := s.gatherUpstreamOutputs(ctx, nodeRun)
	if err != nil {
		slog.Warn("split: failed to gather upstream outputs", "error", err)
		upstreamOutputs = "" // non-fatal
	}

	prompt := s.buildSplitPrompt(issue, upstreamOutputs, cfg)

	// 6. Dispatch agent task for split generation.
	// We use a dedicated phase string "split" and store context on the task.
	task, err := s.dispatchSplitAgentTask(ctx, nodeRun, prompt)
	if err != nil {
		return fmt.Errorf("dispatch split agent: %w", err)
	}

	slog.Info("split: agent task dispatched",
		"node_run_id", util.UUIDToString(nodeRun.ID),
		"task_id", util.UUIDToString(task.ID),
	)
	return nil
}
```

```go
// HandleAgentTaskCompletion is called when the Agent finishes generating
// the split plan. It parses the output, inserts split_task rows, and
// transitions the node run to awaiting_split_review.
func (s *SplitOrchestrator) HandleAgentTaskCompletion(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
	rawOutput json.RawMessage,
) error {
	// 1. Parse Agent output.
	var output AgentSplitOutput
	if err := json.Unmarshal(rawOutput, &output); err != nil {
		slog.Error("split: failed to parse agent output", "error", err)
		s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusFailed)
		return fmt.Errorf("parse agent output: %w", err)
	}
	if len(output.Tasks) == 0 {
		slog.Error("split: agent returned zero tasks")
		s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusFailed)
		return fmt.Errorf("agent returned zero tasks")
	}

	// 2. Insert split_task rows in a transaction.
	run, err := s.Queries.GetWorkflowRun(ctx, nodeRun.WorkflowRunID)
	if err != nil {
		return fmt.Errorf("get run: %w", err)
	}

	workspaceID := run.WorkspaceID

	type insertedTask struct {
		ID    pgtype.UUID
		Index int
	}
	var inserted []insertedTask

	err = s.runInTx(ctx, func(qtx *db.Queries) error {
		for i, t := range output.Tasks {
			dependsOn := []byte("[]")
			if len(t.DependsOnIndices) > 0 {
				dependsOn, _ = json.Marshal(t.DependsOnIndices) // store indices temporarily
			}
			task, err := qtx.CreateSplitTask(ctx, db.CreateSplitTaskParams{
				NodeRunID:              nodeRun.ID,
				WorkspaceID:            workspaceID,
				Title:                  t.Title,
				Description:            nonNullTextString(t.Description),
				SuggestedAssigneeType:  pgtype.Text{String: t.AssigneeType, Valid: t.AssigneeType != ""},
				SuggestedAssigneeID:    stringToUUID(t.AssigneeID),
				DependsOn:              dependsOn,
				SortOrder:              int32(i),
				Status:                 SplitTaskStatusDraft,
			})
			if err != nil {
				return fmt.Errorf("insert split task %d: %w", i, err)
			}
			inserted = append(inserted, insertedTask{ID: task.ID, Index: i})
		}

		// 3. Resolve depends_on_indices → depends_on UUIDs.
		indexToUUID := make(map[int]pgtype.UUID)
		for _, it := range inserted {
			indexToUUID[it.Index] = it.ID
		}
		for i, t := range output.Tasks {
			if len(t.DependsOnIndices) == 0 {
				continue
			}
			deps := make([]string, 0, len(t.DependsOnIndices))
			for _, idx := range t.DependsOnIndices {
				if depID, ok := indexToUUID[idx]; ok {
					deps = append(deps, util.UUIDToString(depID))
				}
			}
			depJSON, _ := json.Marshal(deps)
			_, err := qtx.UpdateSplitTaskFields(ctx, db.UpdateSplitTaskFieldsParams{
				ID:        inserted[i].ID,
				DependsOn: depJSON,
			})
			if err != nil {
				return fmt.Errorf("update depends_on for task %d: %w", i, err)
			}
		}
		return nil
	})
	if err != nil {
		s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusFailed)
		return err
	}

	// 4. Check for DAG cycles.
	tasks, err := s.Queries.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
	if err != nil {
		s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusFailed)
		return fmt.Errorf("list split tasks: %w", err)
	}
	if cycle := s.detectSplitTaskCycle(tasks); cycle != "" {
		slog.Error("split: cycle detected", "cycle", cycle)
		// Agent retry with an error hint is out of scope for this phase.
		s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusFailed)
		return fmt.Errorf("cycle detected in split tasks: %s", cycle)
	}

	// 5. Transition to awaiting_split_review.
	_, err = s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusAwaitingSplitReview)
	return err
}
```

```go
func (s *SplitOrchestrator) ApproveSplit(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
	payload ApproveSplitPayload,
) error {
	// 1. Load run and node metadata.
	run, err := s.Queries.GetWorkflowRun(ctx, nodeRun.WorkflowRunID)
	if err != nil {
		return fmt.Errorf("get run: %w", err)
	}
	_, err = s.Queries.GetWorkflow(ctx, run.WorkflowID)
	if err != nil {
		return fmt.Errorf("get workflow: %w", err)
	}
	node, err := s.Queries.GetWorkflowNode(ctx, nodeRun.WorkflowNodeID)
	if err != nil {
		return fmt.Errorf("get node: %w", err)
	}

	var formatSchema struct {
		SplitConfig SplitConfig `json:"split_config"`
	}
	_ = json.Unmarshal(node.FormatSchema, &formatSchema)
	cfg := formatSchema.SplitConfig

	// 2. Execute all modifications in a single transaction.
	if err := s.runInTx(ctx, func(qtx *db.Queries) error {
		if _, err := qtx.LockSplitNodeRun(ctx, nodeRun.ID); err != nil {
			return fmt.Errorf("lock node run: %w", err)
		}

		// 2a. Mark approved and discarded tasks.
		approvedSet := make(map[string]bool, len(payload.ApprovedTaskIDs))
		for _, id := range payload.ApprovedTaskIDs {
			approvedSet[id] = true
		}

		allTasks, err := qtx.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
		if err != nil {
			return fmt.Errorf("list split tasks: %w", err)
		}

		for _, t := range allTasks {
			taskID := util.UUIDToString(t.ID)
			if approvedSet[taskID] {
				if _, err := qtx.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
					ID:     t.ID,
					Status: SplitTaskStatusApproved,
				}); err != nil {
					return fmt.Errorf("approve task %s: %w", taskID, err)
				}
			} else {
				// Tasks not in approved list are discarded.
				if _, err := qtx.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
					ID:     t.ID,
					Status: SplitTaskStatusDiscarded,
				}); err != nil {
					return fmt.Errorf("discard task %s: %w", taskID, err)
				}
			}
		}

		// 2b. Apply modifications (add/delete/edit).
		for _, mod := range payload.Modifications {
			switch mod.Action {
			case "add":
				depJSON, _ := json.Marshal(mod.DependsOn)
				_, err := qtx.CreateSplitTask(ctx, db.CreateSplitTaskParams{
					NodeRunID:              nodeRun.ID,
					WorkspaceID:            run.WorkspaceID,
					Title:                  mod.Title,
					Description:            nonNullTextString(mod.Description),
					SuggestedAssigneeType:  pgtype.Text{String: mod.SuggestedAssigneeType, Valid: mod.SuggestedAssigneeType != ""},
					SuggestedAssigneeID:    stringToUUID(mod.SuggestedAssigneeID),
					DependsOn:              depJSON,
					SortOrder:              int32(len(allTasks)),
					Status:                 SplitTaskStatusApproved,
				})
				if err != nil {
					return fmt.Errorf("add task: %w", err)
				}
			case "delete":
				taskUUID, _ := util.ParseUUID(mod.ID)
				if err := qtx.DeleteSplitTask(ctx, taskUUID); err != nil {
					return fmt.Errorf("delete task %s: %w", mod.ID, err)
				}
			default:
				// Edit existing task fields.
				taskUUID, _ := util.ParseUUID(mod.ID)
				depJSON, _ := json.Marshal(mod.DependsOn)
				_, err := qtx.UpdateSplitTaskFields(ctx, db.UpdateSplitTaskFieldsParams{
					ID:                     taskUUID,
					Title:                  pgtype.Text{String: mod.Title, Valid: mod.Title != ""},
					Description:            pgtype.Text{String: mod.Description, Valid: mod.Description != ""},
					DependsOn:              depJSON,
					SuggestedAssigneeType:  pgtype.Text{String: mod.SuggestedAssigneeType, Valid: mod.SuggestedAssigneeType != ""},
					SuggestedAssigneeID:    stringToUUID(mod.SuggestedAssigneeID),
				})
				if err != nil {
					return fmt.Errorf("edit task %s: %w", mod.ID, err)
				}
			}
		}

		// 2c. Validate DAG cycles on remaining approved tasks.
		approvedTasks, err := qtx.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
		if err != nil {
			return fmt.Errorf("list approved tasks: %w", err)
		}
		// Filter to approved only.
		var dagTasks []db.MulticaWorkflowSplitTask
		for _, t := range approvedTasks {
			if t.Status == SplitTaskStatusApproved {
				dagTasks = append(dagTasks, t)
			}
		}
		if cycle := s.detectSplitTaskCycle(dagTasks); cycle != "" {
			return fmt.Errorf("cycle detected: %s", cycle)
		}

		// 2d. Create sub-issues for each approved task in topological order.
		ordered := s.topologicalSort(dagTasks)
		parentIssue, _ := s.findIssueForRun(ctx, run)

		for _, t := range ordered {
			issueTitle := t.Title
			assigneeType := t.SuggestedAssigneeType.String
			if assigneeType == "" {
				assigneeType = "agent"
			}
			var assigneeID pgtype.UUID
			if t.SuggestedAssigneeID.Valid {
				assigneeID = t.SuggestedAssigneeID
			}

			// inherit project from parent issue if available
			var projectID pgtype.UUID
			if parentIssue != nil && parentIssue.ProjectID.Valid {
				projectID = parentIssue.ProjectID
			}

			number, err := qtx.IncrementIssueCounter(ctx, run.WorkspaceID)
			if err != nil {
				return fmt.Errorf("increment issue counter: %w", err)
			}

			var parentIssueID pgtype.UUID
			if parentIssue != nil {
				parentIssueID = parentIssue.ID
			}

			issue, err := qtx.CreateIssueWithOrigin(ctx, db.CreateIssueWithOriginParams{
				WorkspaceID:   run.WorkspaceID,
				Title:         issueTitle,
				Description:   nonNullTextString(t.Description),
				Status:        "todo",
				Priority:      "medium",
				AssigneeType:  pgtype.Text{String: assigneeType, Valid: true},
				AssigneeID:    assigneeID,
				CreatorType:   "system",
				CreatorID:     run.TriggeredByID,
				ParentIssueID: parentIssueID,
				Position:      0,
				Number:        number,
				ProjectID:     projectID,
				OriginType:    pgtype.Text{String: "workflow_split", Valid: true},
				OriginID:      t.ID,
				WorkflowID:    pgtype.UUID{},
				WorkflowRunID: pgtype.UUID{},
				StageID:       pgtype.UUID{},
			})
			if err != nil {
				return fmt.Errorf("create sub-issue for task %s: %w", util.UUIDToString(t.ID), err)
			}

			// Write back issue_id.
			if err := qtx.UpdateSplitTaskIssueID(ctx, db.UpdateSplitTaskIssueIDParams{
				ID:      t.ID,
				IssueID: issue.ID,
			}); err != nil {
				return fmt.Errorf("write back issue_id: %w", err)
			}
		}

		return nil
	}); err != nil {
		return err
	}
	// Transaction committed successfully at this point.

	// 3. Transition node run to split_active.
	if _, err := s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusSplitActive); err != nil {
		return fmt.Errorf("transition to split_active: %w", err)
	}

	// 4. Pipeline mode: if all approved tasks have sub-issues created, complete immediately.
	if cfg.Mode == "pipeline" {
		slog.Info("split: pipeline mode — completing split node", "node_run_id", util.UUIDToString(nodeRun.ID))
		if _, err := s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusCompleted); err != nil {
			slog.Error("split: pipeline complete transition failed", "error", err)
		}
		// Still schedule ready tasks — the split node is done, but child workflows
		// should still start asynchronously.
	}

	// 5. Schedule ready tasks (start sub-workflow runs for tasks with all deps satisfied).
	if err := s.ScheduleReadyTasks(ctx, nodeRun); err != nil {
		slog.Error("split: ScheduleReadyTasks failed", "error", err)
	}

	return nil
}
```

```go
// ScheduleReadyTasks starts sub-workflow runs for approved tasks whose
// dependencies are all done, respecting max_concurrency.
func (s *SplitOrchestrator) ScheduleReadyTasks(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
) error {
	return s.runInTx(ctx, func(qtx *db.Queries) error {
	// Lock the node run row for serializing concurrent callbacks.
	_, err := qtx.LockSplitNodeRun(ctx, nodeRun.ID)
	if err != nil {
		return fmt.Errorf("lock node run: %w", err)
	}

	node, err := qtx.GetWorkflowNode(ctx, nodeRun.WorkflowNodeID)
	if err != nil {
		return fmt.Errorf("get node: %w", err)
	}
	var formatSchema struct {
		SplitConfig SplitConfig `json:"split_config"`
	}
	_ = json.Unmarshal(node.FormatSchema, &formatSchema)
	cfg := formatSchema.SplitConfig
	if cfg.MaxConcurrency <= 0 {
		cfg.MaxConcurrency = 5
	}

	tasks, err := qtx.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
	if err != nil {
		return fmt.Errorf("list split tasks: %w", err)
	}

	// Count currently running tasks.
	runningCount := int32(0)
	for _, t := range tasks {
		if t.Status == SplitTaskStatusRunning {
			runningCount++
		}
	}

	subTplID, err := util.ParseUUID(cfg.SubTemplateID)
	if err != nil {
		return fmt.Errorf("parse sub_template_id: %w", err)
	}
	subTpl, err := qtx.GetWorkflow(ctx, subTplID)
	if err != nil {
		return fmt.Errorf("get sub-template: %w", err)
	}

	run, err := qtx.GetWorkflowRun(ctx, nodeRun.WorkflowRunID)
	if err != nil {
		return fmt.Errorf("get run: %w", err)
	}

	// Build a map of task statuses for dependency checking.
	taskStatus := make(map[string]string)
	for _, t := range tasks {
		taskStatus[util.UUIDToString(t.ID)] = t.Status
	}

	for _, t := range tasks {
		if t.Status != SplitTaskStatusCreated {
			continue // only start tasks that are materialized but not yet running
		}

		// Check if runningCount < maxConcurrency.
		if runningCount >= cfg.MaxConcurrency {
			break
		}

		// Check all dependencies are done.
		if !s.allDepsDone(t.DependsOn, taskStatus) {
			continue
		}

		// Start sub-workflow run.
		if !t.IssueID.Valid {
			slog.Warn("split: task has no issue_id, skipping", "task_id", util.UUIDToString(t.ID))
			continue
		}

		issue, err := qtx.GetIssue(ctx, t.IssueID)
		if err != nil {
			slog.Error("split: get issue for task failed", "task_id", util.UUIDToString(t.ID), "error", err)
			continue
		}

		// Append completed dependency outputs as context.
		contextDesc := s.buildDependencyContext(t.DependsOn, tasks)
		if contextDesc != "" {
			issue.Description = pgtype.Text{
				String: issue.Description.String + "\n\n---\n**上游任务输出**:\n" + contextDesc,
				Valid:  true,
			}
		}

		subRun, nodeRuns, err := s.WfService.StartRunForIssue(
			ctx, subTpl, issue,
			"system", util.UUIDToString(run.TriggeredByID),
			run.RuntimeID,
		)
		if err != nil {
			slog.Error("split: start sub-workflow failed", "task_id", util.UUIDToString(t.ID), "error", err)
			// Mark task as failed; barrier mode counts toward failure limit.
			qtx.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
				ID:     t.ID,
				Status: SplitTaskStatusFailed,
			})
			continue
		}

		if err := s.createWorkflowSubIssuesForNodeRuns(ctx, qtx, issue, nodeRuns, run.WorkspaceID); err != nil {
			slog.Error("split: create child workflow sub-issues failed", "task_id", util.UUIDToString(t.ID), "error", err)
			qtx.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
				ID:     t.ID,
				Status: SplitTaskStatusFailed,
			})
			continue
		}
		s.WfService.DispatchRootNodeRuns(ctx, subRun.ID)

		// Write back run_id.
		if err := qtx.UpdateSplitTaskRunID(ctx, db.UpdateSplitTaskRunIDParams{
			ID:    t.ID,
			RunID: subRun.ID,
		}); err != nil {
			slog.Error("split: write back run_id failed", "error", err)
		}

		runningCount++
		slog.Info("split: sub-workflow started",
			"task_id", util.UUIDToString(t.ID),
			"sub_run_id", util.UUIDToString(subRun.ID),
		)
	}

	return nil
	})
}

func (s *SplitOrchestrator) allDepsDone(dependsOnJSON []byte, taskStatus map[string]string) bool {

		return false
	}
	for _, depID := range deps {

			return false

	}
	return true
}

func (s *SplitOrchestrator) buildDependencyContext(dependsOnJSON []byte, tasks []db.MulticaWorkflowSplitTask) string {
	var deps []string
	_ = json.Unmarshal(dependsOnJSON, &deps)
	if len(deps) == 0 {
		return ""
	}

	taskMap := make(map[string]db.MulticaWorkflowSplitTask)
	for _, t := range tasks {
		taskMap[util.UUIDToString(t.ID)] = t
	}

	var parts []string
	for _, depID := range deps {
		if t, ok := taskMap[depID]; ok {
			parts = append(parts, fmt.Sprintf("- **%s**: %s", t.Title, truncateText(t.Description, 200)))
		}
	}
	return strings.Join(parts, "\n")
}

func truncateText(s string, maxLen int) string {
	if len(s) <= maxLen {
		return s
	}
	return s[:maxLen] + "..."
}
```

```go
// HandleChildRunStatusChanged is called when a child workflow run's status
// changes. It updates the split_task status, propagates skip to dependents,
// schedules newly-ready tasks, and checks terminal conditions.
func (s *SplitOrchestrator) HandleChildRunStatusChanged(
	ctx context.Context,
	childRun db.MulticaWorkflowRun,
) error {
	// Find the split_task that references this child run.
	// We need a reverse-lookup: find split_task by run_id.
	// Since we don't have a dedicated query for this, we scan via the issue.
	// In practice, this is called from a listener on workflow_run status changes.
	tasks, err := s.Queries.ListSplitTasksByRun(ctx, childRun.ID)
	if err != nil || len(tasks) == 0 {
		return nil // not a split child
	}

	for _, task := range tasks {
		// Map child run status to split task status.
		var newStatus string
		switch childRun.Status {
		case RunStatusCompleted:
			newStatus = SplitTaskStatusDone
		case RunStatusFailed:
			newStatus = SplitTaskStatusFailed
		case RunStatusCancelled:
			newStatus = SplitTaskStatusCancelled
		default:
			continue // not a terminal status
		}

		if _, err := s.Queries.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
			ID:     task.ID,
			Status: newStatus,
		}); err != nil {
			slog.Error("split: update task status failed", "error", err)
			continue
		}

		// Skip dependent tasks if this one failed or was cancelled.
		if newStatus == SplitTaskStatusFailed || newStatus == SplitTaskStatusCancelled {
			s.skipDependentTasks(ctx, task.NodeRunID, task.ID)
		}

		// Get the parent node run and check terminal conditions.
		nodeRun, err := s.Queries.GetWorkflowNodeRun(ctx, task.NodeRunID)
		if err != nil {
			slog.Error("split: get node run failed", "error", err)
			continue
		}

		// Schedule newly ready tasks.
		if err := s.ScheduleReadyTasks(ctx, nodeRun); err != nil {
			slog.Error("split: ScheduleReadyTasks failed", "error", err)
		}

		// Check terminal conditions (barrier mode only).
		if err := s.checkSplitTerminal(ctx, nodeRun); err != nil {
			slog.Error("split: checkSplitTerminal failed", "error", err)
		}
	}

	return nil
}

// checkSplitTerminal checks if the split node should transition to completed or failed.
func (s *SplitOrchestrator) checkSplitTerminal(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
) error {
	node, err := s.Queries.GetWorkflowNode(ctx, nodeRun.WorkflowNodeID)
	if err != nil {
		return fmt.Errorf("get node: %w", err)
	}
	var formatSchema struct {
		SplitConfig SplitConfig `json:"split_config"`
	}
	_ = json.Unmarshal(node.FormatSchema, &formatSchema)
	cfg := formatSchema.SplitConfig

	// Pipeline mode: already completed in ApproveSplit, no further check needed.
	if cfg.Mode == "pipeline" {
		return nil
	}

	tasks, err := s.Queries.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
	if err != nil {
		return fmt.Errorf("list tasks: %w", err)
	}

	total := 0
	failed := int32(0)
	terminal := int32(0)

	for _, t := range tasks {
		if t.Status == SplitTaskStatusDiscarded {
			continue
		}
		total++
		switch t.Status {
		case SplitTaskStatusDone, SplitTaskStatusFailed, SplitTaskStatusCancelled, SplitTaskStatusSkipped:
			terminal++
		}
		if t.Status == SplitTaskStatusFailed {
			failed++
		}
	}

	// All non-discarded tasks are terminal?
	if terminal < int32(total) {
		return nil // still in progress
	}

	// All terminal: check failure threshold.
	if failed > cfg.MaxFailures {
		slog.Info("split: barrier failed",
			"node_run_id", util.UUIDToString(nodeRun.ID),
			"failed", failed, "max_failures", cfg.MaxFailures,
		)
		_, err := s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusFailed)
		return err
	}

	slog.Info("split: barrier completed",
		"node_run_id", util.UUIDToString(nodeRun.ID),
		"total", total, "failed", failed,
	)
	_, err = s.WfService.TransitionNodeRun(ctx, nodeRun, NodeRunStatusCompleted)
	return err
}

// skipDependentTasks marks tasks that depend on a failed/cancelled task as skipped.
func (s *SplitOrchestrator) skipDependentTasks(
	ctx context.Context,
	nodeRunID pgtype.UUID,
	failedTaskID pgtype.UUID,
) {
	tasks, err := s.Queries.ListSplitTasksByNodeRun(ctx, nodeRunID)
	if err != nil {
		return
	}
	failedIDStr := util.UUIDToString(failedTaskID)
	for _, t := range tasks {
		if t.Status != SplitTaskStatusCreated && t.Status != SplitTaskStatusApproved {
			continue
		}
		var deps []string
		_ = json.Unmarshal(t.DependsOn, &deps)
		for _, dep := range deps {
			if dep == failedIDStr {
				s.Queries.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
					ID:     t.ID,
					Status: SplitTaskStatusSkipped,
				})
				slog.Info("split: skipping dependent task",
					"task_id", util.UUIDToString(t.ID),
					"depends_on", failedIDStr,
				)
				break
			}
		}
	}
}
```

```go
// cancelSplitNode handles the cancel action for a split node.
func (s *SplitOrchestrator) cancelSplitNode(
	ctx context.Context,
	nodeRun db.MulticaWorkflowNodeRun,
) error {
	tasks, err := s.Queries.ListSplitTasksByNodeRun(ctx, nodeRun.ID)
	if err != nil {
		return fmt.Errorf("list split tasks: %w", err)
	}
	run, err := s.Queries.GetWorkflowRun(ctx, nodeRun.WorkflowRunID)
	if err != nil {
		return fmt.Errorf("get run: %w", err)
	}

	for _, t := range tasks {
		// Cancel child workflow runs.
		if t.RunID.Valid {
			if err := s.WfService.CancelRun(ctx, t.RunID); err != nil {
				slog.Warn("split: cancel child run failed",
					"task_id", util.UUIDToString(t.ID),
					"run_id", util.UUIDToString(t.RunID),
					"error", err,
				)
			}
		}
		// Cancel child issues.
		if t.IssueID.Valid {
			s.Queries.UpdateIssueStatus(ctx, db.UpdateIssueStatusParams{
				ID:          t.IssueID,
				Status:      "cancelled",
				WorkspaceID: run.WorkspaceID,
			})
		}
		// Mark non-terminal split tasks as cancelled.
		if !isTerminalSplitTaskStatus(t.Status) {
			s.Queries.UpdateSplitTaskStatus(ctx, db.UpdateSplitTaskStatusParams{
				ID:     t.ID,
				Status: SplitTaskStatusCancelled,
			})
		}
	}

	return nil
}

func isTerminalSplitTaskStatus(s string) bool {
	return s == SplitTaskStatusDone || s == SplitTaskStatusFailed ||
		s == SplitTaskStatusCancelled || s == SplitTaskStatusSkipped ||
		s == SplitTaskStatusDiscarded
}

func (s *SplitOrchestrator) failAllPendingTasks(ctx context.Context, nodeRunID pgtype.UUID) error {

	if err != nil {
		return err
	}
```

// detectSplitTaskCycle runs DFS on the split tasks' depends_on graph.
// Returns a cycle description string, or "" if acyclic.
func (s *SplitOrchestrator) detectSplitTaskCycle(tasks []db.MulticaWorkflowSplitTask) string {
	adj := make(map[string][]string)
	for _, t := range tasks {
		tid := util.UUIDToString(t.ID)
		var deps []string
		json.Unmarshal(t.DependsOn, &deps)
		adj[tid] = deps
	}

	visited := make(map[string]bool)
	onStack := make(map[string]bool)

	var dfs func(node string) string
	dfs = func(node string) string {
		visited[node] = true
		onStack[node] = true
		for _, neighbor := range adj[node] {
			if onStack[neighbor] {
				return fmt.Sprintf("cycle involving %s → %s", node, neighbor)
			}
			if !visited[neighbor] {
				if cycle := dfs(neighbor); cycle != "" {
					return cycle
				}
			}
		}
		onStack[node] = false
		return ""
	}

	for _, t := range tasks {
		tid := util.UUIDToString(t.ID)
		if !visited[tid] {
			if cycle := dfs(tid); cycle != "" {
				return cycle
			}
		}
	}
	return ""
}

// topologicalSort returns tasks in topological order based on depends_on.
func (s *SplitOrchestrator) topologicalSort(tasks []db.MulticaWorkflowSplitTask) []db.MulticaWorkflowSplitTask {
	if len(tasks) <= 1 {
		return tasks
	}

	inDegree := make(map[string]int)
	adj := make(map[string][]string)
	idToTask := make(map[string]db.MulticaWorkflowSplitTask)

	for _, t := range tasks {
		tid := util.UUIDToString(t.ID)
		idToTask[tid] = t
		if inDegree[tid] == 0 {
			_ = inDegree[tid] // ensure key exists
		}
		var deps []string
		json.Unmarshal(t.DependsOn, &deps)
		inDegree[tid] = len(deps)
		for _, dep := range deps {
			adj[dep] = append(adj[dep], tid)
		}
	}

	var queue []string
	for _, t := range tasks {
		tid := util.UUIDToString(t.ID)
		if inDegree[tid] == 0 {
			queue = append(queue, tid)
		}
	}

	var result []db.MulticaWorkflowSplitTask
	for len(queue) > 0 {
		node := queue[0]
		queue = queue[1:]
		result = append(result, idToTask[node])
		for _, neighbor := range adj[node] {
			inDegree[neighbor]--
			if inDegree[neighbor] == 0 {
				queue = append(queue, neighbor)
			}
		}
	}
	return result
}

```typescript
function isSplitNode(node: WorkflowNode): boolean {
  return parseNodeFormat(node.format_schema).kind === "split";
}

function getSplitConfig(node: WorkflowNode): {
  sub_template_id?: string;
  mode?: string;
  max_concurrency?: number;
  max_failures?: number;
} | null {
  if (!node.format_schema || typeof node.format_schema !== "object") {
    return null;
  }
  const schema = node.format_schema as Record<string, unknown>;
  const cfg = schema.split_config;
  if (!cfg || typeof cfg !== "object") return null;
  return cfg as Record<string, unknown>;
}

export function checkSplitTemplateMissing(nodes: WorkflowNode[]): PreflightIssue[] {
  return nodes
    .filter((n) => isSplitNode(n))
    .filter((n) => {
      const cfg = getSplitConfig(n);
      return !cfg || !cfg.sub_template_id;
    })
    .map((n) => ({
      checkId: "split-template-missing" as const,
      severity: "error" as const,
      blocking: true,
      nodeId: n.id,
      nodeTitle: n.title,
      message: "Split node must have a sub-template configured",
    }));
}

/** Check nesting: no node in any sub-template may be kind=split. */
export function checkSplitTemplateNested(
  nodes: WorkflowNode[],
  templateNodesMap: Map<string, WorkflowNode[]>,
): PreflightIssue[] {
  return nodes
    .filter((n) => isSplitNode(n))
    .flatMap((n) => {
      const cfg = getSplitConfig(n);
      if (!cfg?.sub_template_id) return [];
      const subNodes = templateNodesMap.get(cfg.sub_template_id) ?? [];
      const nestedSplit = subNodes.filter(
        (sn) => parseNodeFormat(sn.format_schema).kind === "split",
      );
      if (nestedSplit.length === 0) return [];
      return [
        {
          checkId: "split-template-nested" as const,
          severity: "error" as const,
          blocking: true,
          nodeId: n.id,
          nodeTitle: n.title,
          message: `Sub-template contains a split node — nesting beyond 2 levels is not allowed`,
          detail: nestedSplit.map((ns) => `"${ns.title}"`).join(", "),
        },
      ];
    });
}

export function checkSplitTemplateInactive(
  nodes: WorkflowNode[],
  templateStatuses: Map<string, string>,
): PreflightIssue[] {
  return nodes
    .filter((n) => isSplitNode(n))
    .flatMap((n) => {
      const cfg = getSplitConfig(n);
      if (!cfg?.sub_template_id) return [];
      const status = templateStatuses.get(cfg.sub_template_id);
      if (status === "active" || !status) return [];
      return [
        {
          checkId: "split-template-inactive" as const,
          severity: "error" as const,
          blocking: true,
          nodeId: n.id,
          nodeTitle: n.title,
          message: `Sub-template is ${status} — must be active`,
        },
      ];
    });
}
```

```tsx
import { useEffect, useState, useCallback } from "react";
import { SplitTaskList } from "./split-task-list";
import { SplitTaskDAG } from "./split-task-dag";
import { useSplitTasks, useApproveSplit } from "@multica/core/workflows/queries";
import type { SplitTask, ApproveSplitRequest, SplitTaskModification } from "@multica/core/types";
import { Button } from "@multica/ui/components/ui/button";

export interface SplitReviewPanelProps {
  nodeRunId: string;
  wsId: string;
  workflowId: string;
  runId: string;
  mode: "barrier" | "pipeline";
  maxConcurrency: number;
  onClose: () => void;
}

export function SplitReviewPanel({
  nodeRunId,
  wsId,
  workflowId,
  runId,
  mode,
  maxConcurrency,
  onClose,
}: SplitReviewPanelProps) {
  const { data, isLoading } = useSplitTasks(nodeRunId);
  const approveMutation = useApproveSplit();

  const tasks: SplitTask[] = data?.tasks ?? [];
  const activeTasks = tasks.filter((t) => t.status !== "discarded");

  const [approvedIds, setApprovedIds] = useState<Set<string>>(() => {
    // By default, approve all non-discarded tasks.
    return new Set(activeTasks.map((t) => t.id));
  });
  const [modifications, setModifications] = useState<SplitTaskModification[]>([]);

  useEffect(() => {
    if (activeTasks.length === 0) return;
    setApprovedIds((prev) => {
      if (prev.size > 0) return prev;
      return new Set(activeTasks.map((t) => t.id));
    });
  }, [activeTasks]);

  const handleToggleApprove = useCallback((taskId: string) => {
    setApprovedIds((prev) => {
      const next = new Set(prev);
      if (next.has(taskId)) {
        next.delete(taskId);
      } else {
        next.add(taskId);
      }
      return next;
    });
  }, []);

  const handleEditTask = useCallback((taskId: string, field: string, value: unknown) => {
    setModifications((prev) => {
      const existing = prev.find((m) => "id" in m && m.id === taskId && !("action" in m));
      if (existing && "id" in existing) {
        return prev.map((m) =>
          "id" in m && m.id === taskId
            ? { ...m, [field]: value }
            : m,
        );
      }
      return [...prev, { id: taskId, [field]: value }];
    });
  }, []);

  const handleDeleteTask = useCallback((taskId: string) => {
    setModifications((prev) => [...prev, { action: "delete" as const, id: taskId }]);
    setApprovedIds((prev) => {
      const next = new Set(prev);
      next.delete(taskId);
      return next;
    });
  }, []);

  const handleAddTask = useCallback(() => {
    // Open a simple prompt for title; in production, this would be a more
    // elaborate inline editor. For the plan, use a basic approach.
    const title = prompt("子任务标题:");
    if (!title) return;
    const description = prompt("子任务描述 (可选):") ?? "";
    setModifications((prev) => [
      ...prev,
      {
        action: "add" as const,
        title,
        description,
        depends_on: [],
        suggested_assignee_type: "agent",
        suggested_assignee_id: "",
      },
    ]);
  }, []);

  const handleConfirm = useCallback(async () => {
    const payload: ApproveSplitRequest = {
      approved_task_ids: Array.from(approvedIds),
      modifications,
    };

    try {
      await approveMutation.mutateAsync({ nodeRunId, wsId, workflowId, runId, payload });
      onClose();
    } catch (err) {
      console.error("approve split failed", err);
    }
  }, [approvedIds, modifications, nodeRunId, wsId, workflowId, runId, approveMutation, onClose]);

  if (isLoading) {
    return <div className="p-4 text-sm text-muted-foreground">加载中...</div>;
  }

  return (
    <div className="flex flex-col gap-4 p-4">
      <div className="flex items-center justify-between">
        <h3 className="font-medium">任务拆分审核</h3>
        <span className="text-xs text-muted-foreground">
          模式: {mode === "barrier" ? "barrier" : "pipeline"} · 并发上限: {maxConcurrency}
        </span>
      </div>

      <div className="text-xs text-muted-foreground">
        子任务列表
      </div>

      <SplitTaskList
        tasks={activeTasks}
        approvedIds={approvedIds}
        onToggleApprove={handleToggleApprove}
        onEditTask={handleEditTask}
        onDeleteTask={handleDeleteTask}
        onAddTask={handleAddTask}
      />

      {activeTasks.length > 0 && (
        <>
          <div className="text-xs text-muted-foreground">
            子任务依赖关系
          </div>
          <SplitTaskDAG tasks={activeTasks.filter((t) => approvedIds.has(t.id))} />
        </>
      )}

      <div className="flex items-center justify-between pt-2 border-t">
        <span className="text-xs text-muted-foreground">
          已选择 {approvedIds.size} 个子任务
        </span>
        <div className="flex gap-2">
          <Button variant="outline" size="sm" onClick={onClose}>
            取消
          </Button>
          <Button
            size="sm"
            onClick={handleConfirm}
            disabled={approvedIds.size === 0 || approveMutation.isPending}
          >
            {approveMutation.isPending ? "创建中..." : `确认创建 (${approvedIds.size}个)`}
          </Button>
        </div>
      </div>
    </div>
  );
}
```

```typescript
    if (result.success) {
      expect(result.data.tasks).toHaveLength(1);
      expect(result.data.tasks[0]!.title).toBe("Task 1");
    }
  });

  it("falls back to empty on missing tasks field", () => {
    const raw = { unknown_field: true };
    // .loose() lets this through, but tasks defaults to []
    const result = SplitTaskListSchema.safeParse(raw);
    expect(result.success).toBe(true);
    if (result.success) {
      expect(result.data.tasks).toEqual([]);
    }
  });

  it("falls back on malformed task (missing id)", () => {
    const raw = {
      tasks: [{ title: "no id" }],
    };
    const result = SplitTaskListSchema.safeParse(raw);
    // This will fail because `id: z.string()` is required and missing
    expect(result.success).toBe(false);
    // parseWithFallback at call site returns EMPTY_SPLIT_TASKS_RESPONSE
    expect(EMPTY_SPLIT_TASKS_RESPONSE.tasks).toEqual([]);
  });

  it("survives null array", () => {
    const raw = { tasks: null };
    const result = SplitTaskListSchema.safeParse(raw);
    // null doesn't match z.array(...).default([]), zod coerces
    // z.array() doesn't accept null by default; this tests our defensive design
    expect(result.success).toBe(false);
  });
});
```
