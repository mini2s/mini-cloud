# Dynamic Split Child Workflow Implementation Plan

**Goal:** Move dynamic split execution workflow binding from the split node config to each split task draft, then materialize and schedule every child issue with its own `workflow_id`.

**Architecture:** The split node stores only `split_config.default_issue_workflow_id`; every non-discarded `multica_workflow_split_task` row stores `workflow_id` and `version`. Backend draft APIs fill, validate, patch, approve, and schedule by task-level workflow; frontend APIs and review UI expose deterministic workflow selection while `/split/chat` remains limited to non-workflow fields.

**Tech Stack:** Go 1.26.1, Chi, pgx/sqlc, PostgreSQL migrations, TypeScript strict mode, TanStack Query, Zustand-free shared views, zod response schemas, Vitest, Go `testing`, Playwright E2E.

## Global Constraints

- 本设计替代旧的 `child_workflow_id` 语义，不做旧字段兼容。
- 第一阶段只支持 workflow，不支持 agent、member、squad 或 role 作为子 issue 执行方式。
- Planner agent 不生成、不建议、不选择 `workflow_id`。
- Planner agent 的草案请求如果包含 `workflow_id`，后端直接忽略该字段，不校验、不报错、不写入。
- 自然语言交互不得修改 `workflow_id`；workflow 修改必须走确定性的逐条选择或批量 PATCH API。
- 不在 `split_active` 期间动态新增子 issue。
- 第一阶段单次 approve 的非 discarded task 上限为 50，超过时返回 `422 split_task_limit_exceeded`。
- `default_issue_workflow_id` 必须指向同 workspace 下 active workflow，不能是当前 workflow，且指向的 workflow 不能包含 split 节点。
- 每条非 discarded 草案在 approve 前必须有有效 `workflow_id`。
- `workflow_split` 子 issue 创建后固定为 `assignee_type = "workflow"`、`assignee_id = workflow_id`、`workflow_id = workflow_id`。
- `max_concurrency` 可在审核期和 `split_active` 期间修改；调小不取消已经 running 的子任务。
- `barrier` 模式下父 split node 只有在所有非 discarded 子任务进入终态后才释放下游。
- `pipeline` 模式下父 workflow 在 split group 被 SplitOrchestrator initial dispatch loop 接管后继续。
- 旧 `child_workflow_id` 语义不再被读取或兼容；`default_child_workflow_id` 不作为过渡字段引入。
- 不新增 shared package 依赖，除非已在 `pnpm-workspace.yaml` catalog 中声明。
- `packages/core/` 不使用 `react-dom`、`localStorage`、`process.env` 或 UI 库。
- `packages/views/` 不导入 `next/*` 或 `react-router-dom`。
- 默认只运行相关模块测试；最终集成前按任务风险运行 Go handler/service、core、views 和 E2E 定向测试。

```sql
-- name: UpdateSplitTaskDraftFields :one
UPDATE multica_workflow_split_task
SET title = COALESCE(sqlc.narg('title'), title),
    description = COALESCE(sqlc.narg('description'), description),
    workflow_id = COALESCE(sqlc.narg('workflow_id'), workflow_id),
    depends_on = COALESCE(sqlc.narg('depends_on'), depends_on),

      WHEN sqlc.narg('discarded')::boolean IS TRUE THEN 'discarded'
      WHEN sqlc.narg('discarded')::boolean IS FALSE AND status = 'discarded' THEN 'draft'
      ELSE status
```

```go
func parseSplitConfig(raw []byte) (SplitConfig, error) {
	var format splitNodeFormat
	if len(raw) > 0 {
		if err := json.Unmarshal(raw, &format); err != nil {
			return SplitConfig{}, err
		}
	}
	cfg := format.SplitConfig
	if cfg.Mode == "" {
		cfg.Mode = SplitModeBarrier
	}
	if cfg.Mode != SplitModeBarrier && cfg.Mode != SplitModePipeline {
		return SplitConfig{}, fmt.Errorf("split mode must be barrier or pipeline")
	}
	if cfg.MaxConcurrency < 1 {
		cfg.MaxConcurrency = 5
	}
	if cfg.MaxFailures < 0 {
		cfg.MaxFailures = 0
	}
	cfg.DefaultIssueWorkflowID = strings.TrimSpace(cfg.DefaultIssueWorkflowID)
	if cfg.DefaultIssueWorkflowID == "" {
		return SplitConfig{}, fmt.Errorf("split node is missing default_issue_workflow_id")
	}
	return cfg, nil
}
```

```go
func (s *SplitOrchestrator) validateIssueWorkflow(ctx context.Context, workflowIDText string, parentWorkflowID, workspaceID pgtype.UUID) error {
	workflowID, err := util.ParseUUID(workflowIDText)
	if err != nil {
		return fmt.Errorf("invalid_split_task_workflow: %w", err)
	}
	if workflowID == parentWorkflowID {
		return fmt.Errorf("invalid_split_task_workflow: execution workflow cannot be the parent workflow")
	}
	workflow, err := s.Queries.GetWorkflow(ctx, workflowID)
	if err != nil {
		return fmt.Errorf("invalid_split_task_workflow: get workflow: %w", err)
	}
	if workflow.WorkspaceID != workspaceID {
		return fmt.Errorf("invalid_split_task_workflow: workflow belongs to another workspace")
	}
	if workflow.Status != "active" {
		return fmt.Errorf("invalid_split_task_workflow: workflow is not active")
	}
	nodes, err := s.Queries.ListWorkflowNodes(ctx, workflowID)
	if err != nil {
		return fmt.Errorf("invalid_split_task_workflow: list workflow nodes: %w", err)
	}
	for _, node := range nodes {
		if workflowNodeType(node.FormatSchema) == "split" {
			return fmt.Errorf("invalid_split_task_workflow: execution workflow cannot contain split nodes")
		}
	}
	return nil
}
```

```ts
export function checkSplitDefaultIssueWorkflowConfig(
  nodes: WorkflowNode[],
  splitIssueWorkflows: SplitIssueWorkflowPreflightContext[] = [],
): PreflightIssue[] {
  const workflowsByID = new Map(splitIssueWorkflows.map((workflow) => [workflow.id, workflow]));
  const issues: PreflightIssue[] = [];

  for (const node of nodes) {
    const format = parseNodeFormat(node.format_schema);
    if (format.kind !== "split") continue;

    if (format.split_config?.max_concurrency !== undefined && format.split_config.max_concurrency < 1) {
      issues.push({
        checkId: "split-max-concurrency-invalid",
        severity: "error",
        blocking: true,
        nodeId: node.id,
        nodeTitle: node.title,
        message: "`max_concurrency` must be greater than 0",
      });
    }

    const workflowID = format.split_config?.default_issue_workflow_id;
    if (!workflowID) {
      issues.push({
        checkId: "split-default-issue-workflow-missing",
        severity: "error",
        blocking: true,
        nodeId: node.id,
        nodeTitle: node.title,
        message: "Split node needs a default issue workflow",
      });
      continue;
    }

    const workflow = workflowsByID.get(workflowID);
    if (!workflow) {
      issues.push({
        checkId: "split-default-issue-workflow-invalid",
        severity: "error",
        blocking: true,
        nodeId: node.id,
        nodeTitle: node.title,
        message: "Default issue workflow is invalid",
      });
    }
    if (workflowID === node.workflow_id) {
      issues.push({
        checkId: "split-default-issue-workflow-self",
        severity: "error",
        blocking: true,
        nodeId: node.id,
        nodeTitle: node.title,
        message: "Default issue workflow cannot be the current workflow",
      });
    }
    if (workflow && workflow.status !== "active") {
      issues.push({
        checkId: "split-default-issue-workflow-inactive",
        severity: "error",
        blocking: true,
        nodeId: node.id,
        nodeTitle: node.title,
        message: "Default issue workflow must be active",
      });
    }
    if (workflow?.nodes.some((workflowNode) => parseNodeFormat(workflowNode.format_schema).kind === "split")) {
      issues.push({
        checkId: "split-default-issue-workflow-nested",
        severity: "error",
        blocking: true,
        nodeId: node.id,
        nodeTitle: node.title,
        message: "Default issue workflow cannot contain another split node",
      });
    }
  }

  return issues;
}
```
